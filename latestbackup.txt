    <script type="module">
        import { initializeApp } from "https://www.gstatic.com/firebasejs/11.0.1/firebase-app.js";
        import { getFirestore, doc, setDoc, onSnapshot, serverTimestamp } from "https://www.gstatic.com/firebasejs/11.0.1/firebase-firestore.js";
        import { getAuth, signInAnonymously, onAuthStateChanged } from "https://www.gstatic.com/firebasejs/11.0.1/firebase-auth.js";

        const firebaseConfig = {
            apiKey: "AIzaSyAcvUkTClt50kaJkwBnpU-ZeyqFlflxG10",
            authDomain: "agronauts-ea979.firebaseapp.com",
            projectId: "agronauts-ea979",
            storageBucket: "agronauts-ea979.firebasestorage.app",
            messagingSenderId: "402858856046",
            appId: "1:402858856046:web:cd04957b6ca2d19353607f"
        };

        const app = initializeApp(firebaseConfig);
        const auth = getAuth(app);
        const db = getFirestore(app);

        // This ensures local testing uses the same document as the Arduino
        const actualAppId = "default-app-id";

        const appEl = document.getElementById('app');
        const statusDot = document.getElementById('status-dot');
        const statusText = document.getElementById('status-text');
        const syncText = document.getElementById('sync-text');
        const syncIndicator = document.getElementById('sync-indicator');
        const commandDisplay = document.getElementById('command-display');
        const syncPulse = document.getElementById('sync-pulse');
        const errorLog = document.getElementById('error-log');
        const errorMsg = document.getElementById('error-msg');
        const ruleHelper = document.getElementById('rule-helper');
        const uidTag = document.getElementById('uid-tag');

        appEl.classList.remove('opacity-0');

        async function init() {
            try {
                await signInAnonymously(auth);
            } catch (err) {
                console.error("Auth Error:", err);
                showError(`AUTH FAILED: ${err.message}`);
            }
        }

        function showError(msg, code = "") {
            errorLog.classList.remove('hidden');
            errorMsg.innerText = msg;
            syncText.innerText = "ERROR";
            syncIndicator.className = "w-1.5 h-1.5 rounded-full bg-red-500";
            statusDot.className = "w-2.5 h-2.5 bg-red-500 rounded-full";
            if (code === 'permission-denied') {
                ruleHelper.classList.remove('hidden');
            }
        }

        onAuthStateChanged(auth, (user) => {
            if (user) {
                statusDot.className = "w-2.5 h-2.5 bg-green-500 shadow-[0_0_8px_rgba(34,197,94,0.6)] rounded-full";
                statusText.innerText = "Cloud Authenticated";
                uidTag.innerText = `ID: ${user.uid.substring(0,6)}`;
                startListener();
            }
        });

        function startListener() {
            const commandDocRef = doc(db, 'artifacts', actualAppId, 'public', 'data', 'controls', 'state');
            
            onSnapshot(commandDocRef, (docSnap) => {
                syncIndicator.className = "w-1.5 h-1.5 rounded-full bg-green-500 animate-pulse";
                syncText.innerText = "LIVE SYNC";
                errorLog.classList.add('hidden');
                ruleHelper.classList.add('hidden');

                if (docSnap.exists()) {
                    commandDisplay.innerText = docSnap.data().currentCommand || 'NONE';
                } else {
                    commandDisplay.innerText = 'NONE';
                    setDoc(commandDocRef, { currentCommand: 'NONE', updatedAt: serverTimestamp() }, { merge: true });
                }
            }, (err) => {
                showError(`DB ERROR: ${err.code}`, err.code);
            });
        }

        window.sendAppCommand = async (cmd) => {
            if (!auth.currentUser) return;
            syncPulse.classList.remove('hidden');
            commandDisplay.innerText = cmd;

            try {
                const commandDocRef = doc(db, 'artifacts', actualAppId, 'public', 'data', 'controls', 'state');
                await setDoc(commandDocRef, {
                    currentCommand: cmd,
                    updatedAt: serverTimestamp()
                }, { merge: true });
            } catch (err) {
                showError(`WRITE ERROR: ${err.code}`, err.code);
            } finally {
                setTimeout(() => syncPulse.classList.add('hidden'), 500);
            }
        };

        init();
    </script>
